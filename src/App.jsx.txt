                          }`}
                        >
                          <Truck className="w-4 h-4 shrink-0 text-cyan-400" />
                          <span>{t.methodGermanyDelivery}</span>
                        </button>
                      </div>
                    </div>

                    <div>
                      <label className="block text-[11px] text-slate-400 mb-1">
                        {t.commentLabel}
                      </label>
                      <input
                        type="text"
                        value={checkoutComment}
                        onChange={(e) => setCheckoutComment(e.target.value)}
                        placeholder={t.commentPlaceholder}
                        className="w-full px-3 py-2 bg-slate-900 border border-slate-800 rounded-xl text-xs text-white focus:outline-none focus:border-emerald-500"
                      />
                    </div>
                  </form>
                </>
              )}
            </div>

            {/* Підвал кошика з автоматичним підрахунком знижок та переходом на @zhizha_chaiser */}
            {cartDetails.items.length > 0 && (
              <div className="p-5 bg-slate-950 border-t border-slate-800 space-y-3">
                <div className="space-y-1.5 text-xs">
                  <div className="flex justify-between text-slate-400">
                    <span>{t.totalBottlesLabel}</span>
                    <span className="font-bold text-white">
                      {cartDetails.totalBottles} {t.pcs}
                    </span>
                  </div>

                  {cartDetails.totalDiscountEur > 0 && (
                    <div className="flex justify-between text-slate-400">
                      <span>{t.subtotalLabel}</span>
                      <span className="line-through">€{cartDetails.baseTotalEur.toFixed(2)}</span>
                    </div>
                  )}

                  {cartDetails.bundleDiscountEur > 0 && (
                    <div className="flex justify-between text-emerald-400 font-bold">
                      <span>{t.bundleDiscountLabel}</span>
                      <span>-€{cartDetails.bundleDiscountEur.toFixed(2)}</span>
                    </div>
                  )}

                  <div className="flex justify-between items-baseline pt-2 border-t border-slate-800">
                    <span className="text-sm font-bold text-white">{t.totalToPayLabel}</span>
                    <span className="text-2xl font-black text-emerald-400">
                      €{cartDetails.totalEur.toFixed(2)}
                    </span>
                  </div>
                </div>

                <div className="space-y-2 pt-1">
                  <button
                    type="submit"
                    form="checkout-form"
                    className="w-full py-3.5 px-4 rounded-xl bg-gradient-to-r from-emerald-400 via-teal-400 to-cyan-400 hover:from-emerald-300 hover:to-cyan-300 text-slate-950 font-black text-xs sm:text-sm flex items-center justify-center gap-2 shadow-lg shadow-emerald-500/20 transition"
                  >
                    <Send className="w-4 h-4" />
                    <span>{t.sendToTelegramBtn}</span>
                  </button>

                  <button
                    type="button"
                    onClick={clearCart}
                    className="w-full py-2 rounded-xl bg-slate-900 hover:bg-slate-800 text-slate-400 hover:text-rose-300 text-xs font-semibold flex items-center justify-center gap-1.5 transition"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                    <span>{t.clearCartBtn}</span>
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>
      )}

      {/* =====================================================================
         МОДАЛЬНЕ ВІКНО СФОРМОВАНОЇ НАКЛАДНОЇ (ПЕРЕХІД В TELEGRAM @zhizha_chaiser)
         ===================================================================== */}
      {createdInvoiceModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/85 backdrop-blur-sm">
          <div className="w-full max-w-md rounded-3xl bg-slate-900 border border-emerald-500/40 p-6 shadow-2xl space-y-4">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2.5">
                <div className="w-10 h-10 rounded-2xl bg-emerald-500/20 border border-emerald-500/40 flex items-center justify-center text-emerald-400">
                  <CheckCircle2 className="w-6 h-6" />
                </div>
                <div>
                  <h3 className="text-lg font-black text-white">{t.invoiceModalTitle}</h3>
                  <p className="text-xs text-slate-400">{t.invoiceModalSub}</p>
                </div>
              </div>
              <button
                onClick={() => setCreatedInvoiceModal(null)}
                className="p-2 rounded-xl bg-slate-800 text-slate-400 hover:text-white"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <pre className="p-3.5 rounded-2xl bg-slate-950 border border-slate-800 text-xs text-slate-200 whitespace-pre-wrap font-mono max-h-60 overflow-y-auto">
              {createdInvoiceModal.invoiceText}
            </pre>

            <div className="space-y-2.5">
              <a
                href={createdInvoiceModal.tgWebUrl}
                target="_blank"
                rel="noopener noreferrer"
                className="w-full py-3.5 px-4 rounded-xl bg-gradient-to-r from-sky-400 to-blue-500 hover:from-sky-300 hover:to-blue-400 text-slate-950 font-black text-xs sm:text-sm flex items-center justify-center gap-2 shadow-lg transition"
              >
                <Send className="w-4 h-4" />
                <span>
                  {t.openTgChatBtn} (@{managerTelegram.replace('@', '')})
                </span>
              </a>

              <a
                href={createdInvoiceModal.tgAppUrl}
                className="w-full py-2.5 px-4 rounded-xl bg-slate-800 hover:bg-slate-700 text-sky-300 font-bold text-xs flex items-center justify-center gap-2 transition"
              >
                <span>{t.openTgAppBtn}</span>
              </a>

              <button
                onClick={() => {
                  safeCopyToClipboard(createdInvoiceModal.invoiceText);
                  setCopiedInvoice(true);
                  setTimeout(() => setCopiedInvoice(false), 2000);
                }}
                className="w-full py-2.5 px-4 rounded-xl bg-slate-950 border border-slate-800 text-slate-300 hover:text-white font-bold text-xs flex items-center justify-center gap-2 transition"
              >
                {copiedInvoice ? (
                  <>
                    <Check className="w-4 h-4 text-emerald-400" />
                    <span className="text-emerald-400">{t.copiedText}</span>
                  </>
                ) : (
                  <>
                    <Copy className="w-4 h-4" />
                    <span>{t.copyAgainBtn}</span>
                  </>
                )}
              </button>
            </div>
          </div>
        </div>
      )}

      {/* =====================================================================
         МОДАЛЬНЕ ВІКНО РЕЄСТРАЦІЇ ПОКУПЦЯ (САМОВИВІЗ АБО ДОСТАВКА ПО НІМЕЧЧИНІ)
         ===================================================================== */}
      {isAccountModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/85 backdrop-blur-sm">
          <div className="w-full max-w-lg rounded-3xl bg-slate-900 border border-slate-800 p-6 shadow-2xl space-y-4">
            <div className="flex items-center justify-between border-b border-slate-800 pb-3">
              <div className="flex items-center gap-2.5">
                <User className="w-5 h-5 text-cyan-400" />
                <h3 className="text-lg font-black text-white">{t.accountModalTitle}</h3>
              </div>
              <button
                onClick={() => setIsAccountModalOpen(false)}
                className="p-2 rounded-xl bg-slate-800 text-slate-400 hover:text-white"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            {/* Перемикач вкладок: Реєстрація / Мої накладні */}
            <div className="flex gap-2 bg-slate-950 p-1 rounded-xl border border-slate-800">
              <button
                onClick={() => setAccountTab('profile')}
                className={`flex-1 py-2 rounded-lg text-xs font-bold transition ${
                  accountTab === 'profile'
                    ? 'bg-cyan-500 text-slate-950'
                    : 'text-slate-400 hover:text-white'
                }`}
              >
                {t.tabProfile}
              </button>
              <button
                onClick={() => setAccountTab('orders')}
                className={`flex-1 py-2 rounded-lg text-xs font-bold transition ${
                  accountTab === 'orders'
                    ? 'bg-cyan-500 text-slate-950'
                    : 'text-slate-400 hover:text-white'
                }`}
              >
                {t.tabOrders} ({customerOrders.length})
              </button>
            </div>

            {accountTab === 'profile' ? (
              <form onSubmit={handleSaveCustomerProfile} className="space-y-3 text-xs">
                <p className="text-slate-400 leading-relaxed">{t.regInfoText}</p>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                  <div>
                    <label className="block font-bold text-slate-300 mb-1">
                      {t.yourNameLabel}
                    </label>
                    <input
                      type="text"
                      required
                      value={regName}
                      onChange={(e) => setRegName(e.target.value)}
                      placeholder={t.yourNamePlaceholder}
                      className="w-full px-3 py-2.5 bg-slate-950 border border-slate-800 rounded-xl text-white focus:outline-none focus:border-cyan-500"
                    />
                  </div>
                  <div>
                    <label className="block font-bold text-slate-300 mb-1">
                      {t.yourTgLabel}
                    </label>
                    <input
                      type="text"
                      required
                      value={regTelegram}
                      onChange={(e) => setRegTelegram(e.target.value)}
                      placeholder="@username"
                      className="w-full px-3 py-2.5 bg-slate-950 border border-slate-800 rounded-xl text-white focus:outline-none focus:border-cyan-500"
                    />
                  </div>
                </div>

                <div>
                  <label className="block font-bold text-slate-300 mb-1.5">
                    {t.handoverMethodLabel}
                  </label>
                  <div className="grid grid-cols-1 gap-2">
                    <button
                      type="button"
                      onClick={() => setRegHandover('pickup')}
                      className={`p-3 rounded-xl border text-left text-xs font-bold flex items-center gap-2 transition ${
                        regHandover === 'pickup'
                          ? 'bg-emerald-500/20 border-emerald-400 text-emerald-300'
                          : 'bg-slate-950 border-slate-800 text-slate-300'
                      }`}
                    >
                      <MapPin className="w-4 h-4 text-emerald-400 shrink-0" />
                      <span>{t.methodPickup}</span>
                    </button>

                    <button
                      type="button"
                      onClick={() => setRegHandover('germany_delivery')}
                      className={`p-3 rounded-xl border text-left text-xs font-bold flex items-center gap-2 transition ${
                        regHandover === 'germany_delivery'
                          ? 'bg-cyan-500/20 border-cyan-400 text-cyan-300'
                          : 'bg-slate-950 border-slate-800 text-slate-300'
                      }`}
                    >
                      <Truck className="w-4 h-4 text-cyan-400 shrink-0" />
                      <span>{t.methodGermanyDelivery}</span>
                    </button>
                  </div>
                </div>

                <div className="flex items-center gap-2 pt-2">
                  <button
                    type="submit"
                    className="flex-1 py-3 rounded-xl bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black text-xs sm:text-sm transition"
                  >
                    {t.saveProfileBtn}
                  </button>
                  {customerProfile && (
                    <button
                      type="button"
                      onClick={() => {
                        setCustomerProfile(null);
                        setRegName('');
                        setRegTelegram('');
                        showToast('Профіль покупця очищено');
                      }}
                      className="px-4 py-3 rounded-xl bg-slate-800 text-slate-400 hover:text-rose-300 font-bold text-xs"
                    >
                      {t.logoutBtn}
                    </button>
                  )}
                </div>
              </form>
            ) : (
              <div className="space-y-3 max-h-80 overflow-y-auto pr-1">
                {customerOrders.length === 0 ? (
                  <p className="text-center py-8 text-xs text-slate-400">{t.noOrdersYet}</p>
                ) : (
                  customerOrders.map((ord) => {
                    const cleanManager =
                      managerTelegram.replace('@', '').trim() || 'zhizha_chaiser';
                    const tgLink = `https://t.me/${cleanManager}?text=${encodeURIComponent(
                      ord.invoiceText
                    )}`;
                    return (
                      <div
                        key={ord.id}
                        className="p-3.5 rounded-2xl bg-slate-950 border border-slate-800 space-y-2 text-xs"
                      >
                        <div className="flex items-center justify-between">
                          <span className="font-black text-cyan-400">
                            #{ord.id} • {ord.date}
                          </span>
                          <span className="font-black text-emerald-400 text-sm">
                            €{ord.totalEur.toFixed(2)} ({ord.totalBottles} {t.pcs})
                          </span>
                        </div>
                        <div className="text-slate-300 space-y-0.5">
                          {ord.items.map((it) => (
                            <div key={it.sku}>
                              • {it.line} — {it.name}: {it.qty} шт.
                            </div>
                          ))}
                        </div>
                        <div className="pt-2 border-t border-slate-800 flex items-center justify-between">
                          <span className="text-[11px] text-slate-400">{ord.handover}</span>
                          <a
                            href={tgLink}
                            target="_blank"
                            rel="noopener noreferrer"
                            className="inline-flex items-center gap-1.5 text-sky-400 hover:text-sky-300 font-bold"
                          >
                            <Send className="w-3.5 h-3.5" />
                            <span>{t.reopenInTg}</span>
                          </a>
                        </div>
                      </div>
                    );
                  })
                )}
              </div>
            )}
          </div>
        </div>
      )}

      {/* МОДАЛЬНЕ ВІКНО ВВЕДЕННЯ PIN-КОДУ ВЛАСНИКА (БЕЗ ПІДКАЗКИ ПАРОЛЯ) */}
      {isPinModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/85 backdrop-blur-sm">
          <form
            onSubmit={handleVerifyPin}
            className="w-full max-w-sm rounded-3xl bg-slate-900 border border-violet-500/40 p-6 shadow-2xl space-y-4"
          >
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2">
                <KeyRound className="w-5 h-5 text-violet-400" />
                <h3 className="text-base font-black text-white">Вхід для власника</h3>
              </div>
              <button
                type="button"
                onClick={() => setIsPinModalOpen(false)}
                className="p-1.5 rounded-xl bg-slate-800 text-slate-400 hover:text-white"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <p className="text-xs text-slate-400">
              Введіть персональний PIN-код доступу для перегляду фінансів та управління складом.
            </p>

            <div>
              <input
                type="password"
                autoFocus
                value={pinInput}
                onChange={(e) => {
                  setPinInput(e.target.value);
                  setPinError('');
                }}
                placeholder="••••"
                className="w-full px-4 py-3 bg-slate-950 border border-slate-800 rounded-xl text-center text-lg font-black tracking-widest text-white focus:outline-none focus:border-violet-500"
              />
              {pinError && (
                <p className="text-xs font-bold text-rose-400 text-center mt-2">{pinError}</p>
              )}
            </div>

            <button
              type="submit"
              className="w-full py-3 rounded-xl bg-violet-600 hover:bg-violet-500 text-white font-black text-xs sm:text-sm transition"
            >
              Розблокувати Кабінет Власника
            </button>
          </form>
        </div>
      )}
    </div>
  );
}
